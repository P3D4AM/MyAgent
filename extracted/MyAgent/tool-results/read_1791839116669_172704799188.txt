   126→   126→def _http_json(url: str, headers: Dict[str, str], payload: Dict[str, Any], timeout: int = 180) -> Dict[str, Any]:
   127→   127→    body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
   128→   128→    req = urllib.request.Request(url, data=body, method="POST",
   129→   129→                                 headers={"Content-Type": "application/json",
   130→   130→                                          "User-Agent": "ParsAgent/2.4", **headers})
   131→   131→    opener = _DIRECT if ("127.0.0.1" in url or "localhost" in url) else urllib.request.build_opener()
   132→   132→    try:
   133→   133→        with opener.open(req, timeout=timeout) as r:
   134→   134→            return json.loads(r.read().decode("utf-8"))
   135→   135→    except urllib.error.HTTPError as e:
   136→   136→        try:
   137→   137→            detail = e.read().decode("utf-8", "ignore")[:300]
   138→   138→        except Exception:
   139→   139→            detail = ""
   140→   140→        raise MediaError(f"خطای {e.code} از سرویس تصویرساز{(' — ' + detail) if detail else ''}")
   141→   141→    except urllib.error.URLError as e:
   142→   142→        raise MediaError(f"ارتباط با سرویس تصویرساز برقرار نشد: {e.reason} — اینترنت/VPN را چک کنید")

   161→   161→async def generate_image(ai_cfg: Dict[str, Any], prompt: str, *, provider: str = "zai",
   162→   162→                        model: str = "", size_key: str = "square", out_name: str = "",
   163→   163→                        progress: Optional[Callable[[int, str], None]] = None) -> Dict[str, Any]:
   164→   164→    """یک عکس با AI می‌سازد و در «رسانه‌ها» ذخیره می‌کند.
   165→   165→
   166→   166→    زنجیرهٔ جایگزین: اگر سرویس انتخابی کلید نداشت یا خطا داد، خودکار به
   167→   167→    Pollinations (رایگان و بدون کلید) برمی‌گردد تا کارِ کاربر هرگز نیمه‌کاره نماند.
   168→   168→    """
   169→   169→    prompt = (prompt or "").strip()
   170→   170→    if not prompt:
   171→   171→        raise MediaError("متن توضیح عکس (پرامپت) خالی است")
   172→   172→    size = SIZES.get(size_key, SIZES["square"])
   173→   173→    keys = ai_cfg.get("api_keys") or {}
   174→   174→    tried: List[str] = []
   175→   175→
   176→   176→    def _report(pct: int, msg: str) -> None:
   177→   177→        if progress:
   178→   178→            try:
   179→   179→                progress(pct, msg)
   180→   180→            except Exception:
   181→   181→                pass
   182→   182→
   183→   183→    chain: List[Tuple[str, str]] = []
   184→   184→    if provider and provider in IMAGE_PROVIDERS and provider != "pollinations":
   185→   185→        m = model or IMAGE_PROVIDERS[provider]["models"][0]["id"]
   186→   186→        chain.append((provider, m))
   187→   187→    chain.append(("pollinations", model if provider == "pollinations" else "flux"))
   188→   188→
   189→   189→    data_bytes: Optional[bytes] = None
   190→   190→    used = ""
   191→   191→    for prov, mdl in chain:
   192→   192→        info = IMAGE_PROVIDERS[prov]
   193→   193→        key = keys.get(info.get("key_field", ""), "") if info.get("key_field") else "free"
   194→   194→        if not key:
   195→   195→            tried.append(f"{info['name']}: کلید API ندارد")
   196→   196→            continue
   197→   197→        _report(30, f"در حال ساخت عکس با {info['name']} ({mdl}) …")
   198→   198→        attempts = 2 if prov == "pollinations" else 1  # سرویس رایگان گاهی کند است — یک تلاش دیگر
   199→   199→        for attempt in range(attempts):
   200→   200→            try:
   201→   201→                if prov == "pollinations":
   202→   202→                    q = urllib.parse.quote(prompt[:1500], safe="")
   203→   203→                    url = (f"https://image.pollinations.ai/prompt/{q}"
   204→   204→                           f"?width={size['w']}&height={size['h']}"
   205→   205→                           f"&seed={random.randrange(1, 10**6)}&nologo=true&model={mdl}")
   206→   206→                    data_bytes = await asyncio.to_thread(_download_bytes, url)
   207→   207→                else:
   208→   208→                    payload = {"model": mdl, "prompt": prompt[:3800], "size": size["api"], "n": 1}
   209→   209→                    if mdl == "dall-e-3":
   210→   210→                        payload["response_format"] = "url"
   211→   211→                    res = await asyncio.to_thread(
   212→   212→                        _http_json, info["base"] + "/images/generations",
   213→   213→                        {"Authorization": f"Bearer {key}"}, payload)
   214→   214→                    item = ((res.get("data") or [{}])[0])
   215→   215→                    if item.get("b64_json"):
   216→   216→                        data_bytes = base64.b64decode(item["b64_json"])
   217→   217→                    elif item.get("url"):
   218→   218→                        data_bytes = await asyncio.to_thread(_download_bytes, item["url"])
   219→   219→                if data_bytes:
   220→   220→                    used = f"{info['name']} · {mdl}"
   221→   221→                    break
   222→   222→            except MediaError as e:
   223→   223→                tried.append(f"{info['name']}: {e}")
   224→   224→                data_bytes = None
   225→   225→                continue
   226→   226→            except Exception as e:  # noqa
   227→   227→                tried.append(f"{info['name']}: {e}")
   228→   228→                data_bytes = None
   229→   229→                continue
   230→   230→        if data_bytes:
   231→   231→            break
   232→   232→    if not data_bytes:
   233→   233→        raise MediaError("هیچ سرویس تصویرسازی جواب نداد — " + " | ".join(tried[:3]))
   234→   234→
   235→   235→    # ذخیره در رسانه‌ها (فرمت واقعی را از بایت‌ها تشخیص می‌دهیم)
   236→   236→    kind = "png" if data_bytes[:8] == b"\x89PNG\r\n\x1a\n" else ("jpg" if data_bytes[:3] == b"\xff\xd8\xff" else "png")
   237→   237→    name = out_name or _now_stamped("ai-image", prompt)
   238→   238→    name = re.sub(r"\.(png|jpg|jpeg|webp)$", "", name, flags=re.I) + f".{kind}"
   239→   239→    path = MEDIA_DIR / name
   240→   240→    path.write_bytes(data_bytes)
   241→   241→    _report(100, f"✅ عکس ساخته شد: {name}")
   242→   242→    return {"ok": True, "file": name, "path": str(path), "provider": used,
   243→   243→            "url": f"/media/{urllib.parse.quote(name)}", "size": size["api"]}

   247→   247→async def _llm_scenes(llm_json: Callable, ai_cfg: Dict[str, Any], provider: str, model: str,
   248→   248→                      topic: str, n_scenes: int, lang: str, seconds: int) -> List[Dict[str, Any]]:
   249→   249→    """فیلم‌نامهٔ صحنه‌به‌صحنه — مثل MoneyPrinterTurbo: ماجرای هر صحنه + پرامپت تصویرش."""
   250→   250→    per = max(4, min(20, seconds // max(n_scenes, 1)))
   251→   251→    system = (
   252→   252→        "تو فیلم‌نامه‌نویس ویدیوهای کوتاه شبکه‌های اجتماعی هستی (سبک MoneyPrinterTurbo). "
   253→   253→        f"موضوع را به {n_scenes} صحنهٔ پیوسته تبدیل کن که مجموعاً حدود {seconds} ثانیه ویدیو شود. "
   254→   254→        f"زبان گویندگی: {'فارسی' if lang == 'fa' else 'انگلیسی'}. "
   255→   255→        "هر صحنه فقط یک جملهٔ کوتاه و طبیعی برای «بخواندن» دارد (۶ تا ۱۴ کلمه) و یک image_prompt انگلیسی "
   256→   256→        "برای تصویر همان صحنه (سبک سینمایی، نور خوب، بدون متن داخل تصویر). متن داخل تصویر ممنوع. "
   257→   257→        'خروجی فقط JSON: {"scenes":[{"narration":"...","image_prompt":"..."}]}'
   258→   258→    )
   259→   259→    data = await llm_json(ai_cfg, provider, model, [
   260→   260→        {"role": "system", "content": system},
   261→   261→        {"role": "user", "content": f"موضوع ویدیو: {topic}"},
   262→   262→    ], temperature=0.7)
   263→   263→    scenes = []
   264→   264→    for s in (data.get("scenes") or [])[:n_scenes * 2]:
   265→   265→        nar = str(s.get("narration") or "").strip()
   266→   266→        img = str(s.get("image_prompt") or "").strip()
   267→   267→        if nar:
   268→   268→            scenes.append({"narration": nar, "image_prompt": img or topic})
   269→   269→    if len(scenes) < 2:
   270→   270→        raise MediaError("مدل، فیلم‌نامهٔ معتبری نساخت — دوباره تلاش کنید یا مدل بهتری انتخاب کنید")
   271→   271→    return scenes[:n_scenes]

   274→   274→async def _tts_scene(text: str, voice: str, out_mp3: Path) -> Dict[str, Any]:
   275→   275→    """گویندگی رایگان با edge-tts + جمع‌آوری زمان کلمات (برای زیرنویس دقیق)."""
   276→   276→    try:
   277→   277→        import edge_tts
   278→   278→    except ImportError:
   279→   279→        raise MediaError("کتابخانهٔ edge-tts نصب نیست — در خط فرمان: pip install edge-tts")
   280→   280→    words: List[Dict[str, Any]] = []
   281→   281→    out_mp3.parent.mkdir(parents=True, exist_ok=True)
   282→   282→    with open(out_mp3, "wb") as f:
   283→   283→        # boundary=WordBoundary — زمان دقیق هر کلمه برای زیرنویس (پیش‌فرض نسخهٔ 7+ جمله‌ای است)
   284→   284→        try:
   285→   285→            communicate = edge_tts.Communicate(text, voice, rate="+4%", boundary="WordBoundary")
   286→   286→        except TypeError:  # نسخه‌های قدیمی edge-tts — WordBoundary خودکار می‌آمد
   287→   287→            communicate = edge_tts.Communicate(text, voice, rate="+4%")
   288→   288→        async for chunk in communicate.stream():
   289→   289→            if chunk["type"] == "audio":
   290→   290→                f.write(chunk["data"])
   291→   291→            elif chunk["type"] in ("WordBoundary", "SentenceBoundary"):
   292→   292→                words.append({"t": chunk["offset"] / 1e7, "d": chunk["duration"] / 1e7,
   293→   293→                              "w": chunk.get("text", "")})
   294→   294→    dur = (words[-1]["t"] + words[-1]["d"]) if words else max(3.0, len(text.split()) * 0.42)
   295→   295→    return {"words": words, "duration": round(dur, 3), "file": out_mp3}

   379→   379→def _compose_scene_image(material: Path, out_png: Path, w: int, h: int,
   380→   380→                         narration: str, accent: Tuple[int, int, int]) -> None:
   381→   381→    """تصویر صحنه + نوار زیرنویس فارسی → یک PNG آمادهٔ ویدیو."""
   382→   382→    from PIL import Image, ImageDraw, ImageEnhance
   383→   383→    if material and Path(material).exists():
   384→   384→        img = Image.open(material).convert("RGB")
   385→   385→        # برش مرکزی به نسبت هدف
   386→   386→        sr, tr = img.width / img.height, w / h
   387→   387→        if sr > tr:
   388→   388→            nw = int(img.height * tr)
   389→   389→            img = img.crop(((img.width - nw) // 2, 0, (img.width - nw) // 2 + nw, img.height))
   390→   390→        else:
   391→   391→            nh = int(img.width / tr)
   392→   392→            img = img.crop((0, (img.height - nh) // 2, img.width, (img.height - nh) // 2 + nh))
   393→   393→        img = img.resize((w, h), Image.LANCZOS)
   394→   394→        img = ImageEnhance.Color(img).enhance(1.08)
   395→   395→        canvas = img
   396→   396→    else:
   397→   397→        # پس‌زمینهٔ گرادیان — وقتی تصویر صحنه در دسترس نیست، صحنه هنوز قابل‌قبول بماند
   398→   398→        canvas = Image.new("RGB", (w, h))
   399→   399→        px = canvas.load()
   400→   400→        for y in range(h):
   401→   401→            t = y / max(1, h - 1)
   402→   402→            r = int(14 + (accent[0] // 12) * (1 - t))
   403→   403→            g = int(18 + (accent[1] // 12) * (1 - t))
   404→   404→            b = int(24 + (accent[2] // 12) * (1 - t))
   405→   405→            for x in range(0, w, 16):
   406→   406→                for dx in range(min(16, w - x)):
   407→   407→                    px[x + dx, y] = (r, g, b)
   408→   408→    draw = ImageDraw.Draw(canvas, "RGBA")
   409→   409→    # نوار نیمه‌شفاف پایین + متنِ شکل‌دهی‌شدهٔ راست‌به‌چپ
   410→   410→    band_h = max(140, h // 6)
   411→   411→    base = canvas.convert("RGBA")
   412→   412→    overlay = Image.new("RGBA", (w, h), (0, 0, 0, 0))
   413→   413→    overlay.paste(Image.new("RGBA", (w, band_h), (8, 10, 14, 168)), (0, h - band_h))
   414→   414→    canvas = Image.alpha_composite(base, overlay)
   415→   415→    draw = ImageDraw.Draw(canvas, "RGBA")
   416→   416→    font = _load_font(max(34, h // 26))
   417→   417→    # شکستن سطر بر اساس پهنای واقعی — روی متنِ منطقی (راست‌به‌چپ خودش هنگام رندر اعمال می‌شود)
   418→   418→    max_w = int(w * 0.86)
   419→   419→    lines, cur = [], ""
   420→   420→    for word in (narration or "").split(" "):
   421→   421→        trial = (cur + " " + word).strip()
   422→   422→        if _text_width(draw, trial, font) <= max_w or not cur:
   423→   423→            cur = trial
   424→   424→        else:
   425→   425→            lines.append(cur)
   426→   426→            cur = word
   427→   427→    if cur:
   428→   428→        lines.append(cur)
   429→   429→    lines = lines[:3]
   430→   430→    lh = font.size + max(14, h // 90)
   431→   431→    total = lh * len(lines)
   432→   432→    y = h - band_h + (band_h - total) // 2 + h // 200
   433→   433→    for ln in lines:
   434→   434→        tw = _text_width(draw, ln, font)
   435→   435→        _draw_text(draw, ((w - tw) / 2, y), ln, font,
   436→   436→                   fill=(255, 255, 255, 245),
   437→   437→                   stroke_width=max(1, h // 900), stroke_fill=(0, 0, 0, 180))
   438→   438→        y += lh
   439→   439→    # خط تاکید بالای نوار (رنگ برند)
   440→   440→    draw.rectangle([0, h - band_h, w, h - band_h + max(4, h // 350)], fill=(*accent, 255))
   441→   441→    canvas.convert("RGB").save(out_png, "PNG")

   444→   444→def _srt_from_words(words: List[Dict[str, Any]], path: Path) -> None:
   445→   445→    """زیرنویس SRT جداگانه هم ذخیره می‌کنیم (برای پلتفرم‌هایی که فایل کپشن می‌گیرند)."""
   446→   446→    def _t(s: float) -> str:
   447→   447→        hh = int(s // 3600); mm = int(s % 3600 // 60); ss = int(s % 60); ms = int((s - int(s)) * 1000)
   448→   448→        return f"{hh:02d}:{mm:02d}:{ss:02d},{ms:03d}"
   449→   449→    if not words:
   450→   450→        return
   451→   451→    cues, cur = [], []  # cur = لیست (t, d, w)
   452→   452→    for w in words:
   453→   453→        if cur and w["t"] - cur[0][0] > 3.5:
   454→   454→            cues.append((cur[0][0], cur[-1][0] + cur[-1][1], " ".join(x[2] for x in cur)))
   455→   455→            cur = []
   456→   456→        cur.append((w["t"], w["d"], w["w"]))
   457→   457→    if cur:
   458→   458→        cues.append((cur[0][0], cur[-1][0] + cur[-1][1], " ".join(x[2] for x in cur)))
   459→   459→    with open(path, "w", encoding="utf-8") as f:
   460→   460→        for i, (a, b, txt) in enumerate(cues, 1):
   461→   461→            # SRT باید متنِ منطقیِ خام باشد — پلیرها خودشان رسم‌الخط می‌کنند
   462→   462→            f.write(f"{i}\n{_t(a)} --> {_t(b)}\n{txt}\n\n")
