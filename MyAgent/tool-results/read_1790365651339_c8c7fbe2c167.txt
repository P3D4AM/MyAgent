    76→class LogHub:
    77→    """مرکز پخش رویداد به همهٔ کلاینت‌های متصل‌شده با WebSocket."""
    78→
    79→    def __init__(self) -> None:
    80→        self.clients: set = set()
    81→        self.history: deque = deque(maxlen=300)
    82→        self._lock = threading.Lock()
    83→
    84→    def emit(self, name: str, level: str, msg: str) -> None:
    85→        item = {
    86→            "type": "log", "ts": datetime.now().strftime("%H:%M:%S"),
    87→            "name": name, "level": level, "msg": str(msg)[:2000],
    88→        }
    89→        with self._lock:
    90→            self.history.append(item)
    91→        self.push(item)
    92→
    93→    def push(self, item: dict) -> None:
    94→        """پخش یک رویداد (log/job/record) به همه — امن از هر نخ."""
    95→        loop = LOOP
    96→        if loop is None:
    97→            return
    98→        try:
    99→            asyncio.run_coroutine_threadsafe(self._broadcast(item), loop)
   100→        except Exception:
   101→            pass
   102→
   103→    async def _broadcast(self, item: dict) -> None:
   104→        data = json.dumps(item, ensure_ascii=False)
   105→        dead = []
   106→        for ws in list(self.clients):
   107→            try:
   108→                await ws.send_text(data)
   109→            except Exception:
   110→                dead.append(ws)
   111→        for ws in dead:
   112→            self.clients.discard(ws)
   113→
   114→

   201→class AgentWorker:
   202→    """صف تک‌نخمی برای هر عملیات ایجنت (مرورگر، تسک، ورود، …).
   203→
   204→    sync-API پایتونِ Playwright بین‌نخ امن نیست؛ همهٔ کارهای مرورگری از
   205→    این نخ عبور می‌کنند تا هیچ‌وقت دو نخ هم‌زمان کانتکست را لمس نکنند.
   206→    """
   207→
   208→    def __init__(self) -> None:
   209→        self._q: "queue.Queue" = queue.Queue()
   210→        threading.Thread(target=self._loop, name="parsagent-worker",
   211→                         daemon=True).start()
   212→
   213→    def _loop(self) -> None:
   214→        while True:
   215→            fn, args, done, slot = self._q.get()
   216→            try:
   217→                slot["result"] = fn(*args)
   218→            except BaseException as exc:            # noqa: BLE001
   219→                slot["error"] = exc
   220→            finally:
   221→                done.set()
   222→
   223→    def run(self, fn: Callable, *args: Any) -> Any:
   224→        done = threading.Event()
   225→        slot: Dict[str, Any] = {"result": None, "error": None}
   226→        self._q.put((fn, args, done, slot))
   227→        done.wait()
   228→        if slot["error"] is not None:
   229→            raise slot["error"]
   230→        return slot["result"]
   231→

   282→class _Cancelled(Exception):

   285→
   286→JOBS: Dict[str, dict] = {}
   287→_JOB_ORDER: List[str] = []
   288→_JOB_LOCK = threading.Lock()
   289→_job_seq = itertools.count(1)
   290→
   291→
   292→def _job_cancelled(jid: str) -> bool:
   293→    with _JOB_LOCK:
   294→        return bool(JOBS.get(jid, {}).get("cancel"))

   296→
   297→def _job_public(job: dict, include_result: bool = True) -> dict:
   298→    base = {
   299→        "id": job["id"], "kind": job["kind"], "label": job["label"],
   300→        "status": job["status"], "started_at": job["started_at"],
   301→        "finished_at": job["finished_at"], "error": job["error"],
   302→        "cancellable": job["cancellable"],
   303→    }
   304→    if include_result and job.get("result") is not None:
   305→        try:
   306→            s = json.dumps(job["result"], ensure_ascii=False, default=str)
   307→            if len(s) < 4000:
   308→                base["result"] = job["result"]
   309→            else:
   310→                base["result_omitted"] = True
   311→        except Exception:
   312→            base["result_omitted"] = True
   313→    return base
   314→
   315→
   316→def _recent_jobs(n: int = 20) -> List[dict]:
   317→    with _JOB_LOCK:
   318→        return [JOBS[j] for j in _JOB_ORDER[-n:]][::-1]
   319→
   320→
   321→def _new_job(kind: str, label: str, fn: Callable, *args: Any,
   322→             via_worker: bool = True, inject_id: bool = False) -> dict:
   323→    jid = f"j{next(_job_seq)}"
   324→    job = {
   325→        "id": jid, "kind": kind, "label": label, "status": "running",
   326→        "started_at": _now_iso(), "finished_at": None,
   327→        "result": None, "error": None, "cancel": False,
   328→        "cancellable": kind == "macro",
   329→    }
   330→    with _JOB_LOCK:
   331→        JOBS[jid] = job
   332→        _JOB_ORDER.append(jid)
   333→        while len(_JOB_ORDER) > 80:
   334→            JOBS.pop(_JOB_ORDER.pop(0), None)
   335→
   336→    def _run() -> None:
   337→        call_args = (jid,) + args if inject_id else args
   338→        try:
   339→            if via_worker:
   340→                job["result"] = AGENT.run(fn, *call_args)
   341→            else:
   342→                job["result"] = fn(*call_args)
   343→            job["status"] = "cancelled" if job["cancel"] else "done"
   344→        except _Cancelled as exc:
   345→            job["status"] = "cancelled"
   346→            job["error"] = str(exc)
   347→        except Exception as exc:                   # noqa: BLE001
   348→            job["status"] = "failed"
   349→            job["error"] = str(exc)[:600]
   350→            LOG.emit("server", "error", f"کار «{label}» شکست خورد: {str(exc)[:220]}")
   351→        finally:
   352→            job["finished_at"] = _now_iso()
   353→            LOG.push({"type": "job", "job": _job_public(job)})
   354→
   355→    threading.Thread(target=_run, daemon=True, name=f"job-{jid}").start()
   356→    LOG.push({"type": "job", "job": _job_public(job)})
   357→    return job

   360→# --------------------------------------------------------------------------- #
   361→# ۶) ضبط‌کنندهٔ تحت وب — همان فرمت ماکروی InputRecorder (سازگار با CLI)
   362→#    F9 = پایان و ذخیره، F10 = لغو — یا از خود پنل.
   363→# --------------------------------------------------------------------------- #
   364→class WebRecorder:
   365→    def __init__(self) -> None:
   366→        self.events: List[dict] = []
   367→        self.active = False
   368→        self.last_macro: Optional[dict] = None
   369→        self._t0 = 0.0
   370→        self._last_move = 0.0
   371→        self._m = None
   372→        self._k = None
   373→
   374→    def _now(self) -> float:
   375→        return round(time.monotonic() - self._t0, 4)
   376→
   377→    def start(self) -> None:
   378→        if self.active:
   379→            return
   380→        from pynput import keyboard, mouse
   381→        self.events = []
   382→        self.last_macro = None
   383→        self._t0 = time.monotonic()
   384→        self._last_move = 0.0
   385→        self.active = True
   386→        self._m = mouse.Listener(on_move=self._om, on_click=self._oc,
   387→                                 on_scroll=self._os)
   388→        self._k = keyboard.Listener(on_press=self._ok_press,
   389→                                    on_release=self._ok_rel)
   390→        self._m.start()
   391→        self._k.start()
   392→        threading.Thread(target=self._tick, daemon=True).start()
   393→        LOG.emit("recorder", "info", "● ضبط شروع شد — پایان: دکمهٔ پنل یا کلید F9")
   394→
   395→    def _tick(self) -> None:
   396→        while self.active:
   397→            LOG.push({"type": "record", "active": True,
   398→                      "count": len(self.events), "duration": self._now()})
   399→            time.sleep(0.8)
   400→
   401→    def _om(self, x: int, y: int) -> None:
   402→        if not self.active:
   403→            return
   404→        t = time.monotonic()
   405→        if t - self._last_move < 0.025:            # حداکثر ~۴۰ نمونه در ثانیه
   406→            return
   407→        self._last_move = t
   408→        self.events.append({"t": self._now(), "type": "move", "x": x, "y": y})
   409→
   410→    def _oc(self, x: int, y: int, button, pressed: bool) -> None:
   411→        if not self.active:
   412→            return
   413→        self.events.append({"t": self._now(), "type": "click", "x": x, "y": y,
   414→                            "button": button.name, "pressed": pressed})
   415→
   416→    def _os(self, x: int, y: int, dx: int, dy: int) -> None:
   417→        if not self.active:
   418→            return
   419→        self.events.append({"t": self._now(), "type": "scroll",
   420→                            "x": x, "y": y, "dx": dx, "dy": dy})
   421→
   422→    @staticmethod
   423→    def _key_of(key) -> Optional[str]:
   424→        ch = getattr(key, "char", None)
   425→        if ch:
   426→            return ch
   427→        return getattr(key, "name", None) or str(key)
   428→
   429→    def _ok_press(self, key) -> None:
   430→        if not self.active:
   431→            return
   432→        name = getattr(key, "name", None)
   433→        if name in ("f9", "f10"):                   # کلیدهای کنترلی ضبط
   434→            self.stop(save=(name == "f9"))
   435→            return
   436→        self.events.append({"t": self._now(), "type": "key",
   437→                            "key": self._key_of(key), "pressed": True})
   438→
   439→    def _ok_rel(self, key) -> None:
   440→        if not self.active:
   441→            return
   442→        ch = getattr(key, "char", None)
   443→        if ch:                                      # برای حروف، press کافی است
   444→            return
   445→        self.events.append({"t": self._now(), "type": "key",
   446→                            "key": self._key_of(key), "pressed": False})
   447→
   448→    def stop(self, save: bool = True) -> Optional[dict]:
   449→        self.active = False
   450→        for listener in (self._m, self._k):
   451→            try:
   452→                listener.stop()
   453→            except Exception:
   454→                pass
   455→        self._m = self._k = None
   456→        macro = {
   457→            "meta": {
   458→                "version": 1,
   459→                "recorded_at": datetime.now().isoformat(timespec="seconds"),
   460→                "duration": self._now(),
   461→                "events": len(self.events),
   462→            },
   463→            "events": self.events,
   464→        }
   465→        if save and self.events:
   466→            self.last_macro = macro
   467→            LOG.emit("recorder", "info",
   468→                     f"■ ضبط تمام شد — {len(self.events)} رخداد "
   469→                     f"({macro['meta']['duration']:.1f}s)")
   470→        else:
   471→            LOG.emit("recorder", "warning",
   472→                     f"■ ضبط لغو شد — {len(self.events)} رخداد دور ریخته شد")
   473→        LOG.push({"type": "record", "active": False,
   474→                  "count": len(self.events), "duration": macro["meta"]["duration"]})
   475→        return self.last_macro
   476→
   477→    def status(self) -> dict:
   478→        dur = self._now() if self.active else \
   479→            (self.last_macro or {}).get("meta", {}).get("duration", 0)
   480→        return {"active": self.active, "count": len(self.events),
   481→                "duration": round(dur, 1)}
   482→

   510→def _type_text_direct(text: str) -> None:
   511→    """تایپ متن — متن فارسی از طریق کلیپ‌بورد (typewrite فقط ASCII است)."""
   512→    if not text:
   513→        return
   514→    if all(ord(c) < 128 for c in text):
   515→        pyautogui.typewrite(text, interval=0.012)
   516→        return
   517→    if pyperclip is None:
   518→        raise RuntimeError("برای متن فارسی، پکیج pyperclip لازم است")
   519→    pyperclip.copy(text)
   520→    pyautogui.hotkey("ctrl", "v")
   521→    time.sleep(0.05)

   524→def _run_macro(job_id: str, events: List[dict], speed: float, loop: int) -> dict:
   525→    _require_input()
   526→    speed = max(0.1, min(float(speed), 10.0))
   527→    rounds = max(1, int(loop))
   528→    played = 0
   529→    for rnd in range(1, rounds + 1):
   530→        buf = ""
   531→        prev_t = 0.0
   532→        for ev in events:
   533→            if _job_cancelled(job_id):
   534→                raise _Cancelled(f"ماکرو در دور {rnd} لغو شد")
   535→            wait = (ev["t"] - prev_t) / speed
   536→            prev_t = ev["t"]
   537→            if wait > 0:
   538→                time.sleep(min(wait, 30.0))        # حداکثر ۳۰ ثانیه مکث در هر گام
   539→            et = ev.get("type")
   540→            if et == "move":
   541→                if buf:
   542→                    _type_text_direct(buf)
   543→                    buf = ""
   544→                pyautogui.moveTo(ev["x"], ev["y"], duration=0)
   545→            elif et == "click":
   546→                if buf:
   547→                    _type_text_direct(buf)
   548→                    buf = ""
   549→                if ev.get("pressed"):
   550→                    pyautogui.click(x=ev["x"], y=ev["y"],
   551→                                    button=ev.get("button", "left"),
   552→                                    clicks=1, interval=0.05)
   553→            elif et == "scroll":
   554→                if buf:
   555→                    _type_text_direct(buf)
   556→                    buf = ""
   557→                pyautogui.scroll(int(ev.get("dy", 0)))
   558→            elif et == "key":
   559→                key = ev.get("key")
   560→                if not key:
   561→                    continue
   562→                if len(key) == 1:                  # کاراکتر معمولی
   563→                    if ev.get("pressed", True):
   564→                        buf += key
   565→                else:                              # کلید ویژه
   566→                    if buf:
   567→                        _type_text_direct(buf)
   568→                        buf = ""
   569→                    mapped = _PAGUI_KEY.get(key, key)
   570→                    if key in _MODIFIERS:
   571→                        if ev.get("pressed", True):
   572→                            pyautogui.keyDown(mapped)
   573→                        else:
   574→                            pyautogui.keyUp(mapped)
   575→                    elif ev.get("pressed", True):
   576→                        pyautogui.press(mapped)
   577→            played += 1
   578→        if buf:
   579→            _type_text_direct(buf)
   580→    return {"events_played": played, "rounds": rounds, "speed": speed}

   718→def _platform_open(platform: str) -> dict:
   719→    eng = _get_engine()
   720→    if platform not in PLATFORMS:
   721→        raise ValueError(f"پلتفرم ناشناخته: {platform}")
   722→    mod = eng.platform(platform)                    # پروفایل ماندگار + لانچ تنبل
   723→    page = getattr(mod, "_page", None)
   724→    if page is not None:
   725→        try:
   726→            if not page.is_closed():
   727→                page.goto(mod.base_url, wait_until="domcontentloaded",
   728→                          timeout=45_000)
   729→                try:
   730→                    page.bring_to_front()
   731→                except Exception:
   732→                    pass
   733→                return {"already_open": True, "url": page.url}
   734→        except Exception:
   735→            pass
   736→    page = mod.open()                               # صفحهٔ تازه + رفتن به سایت
   737→    return {"already_open": False, "url": page.url,
   738→            "channel": eng._browser.channel}

   799→            names = [t.get("name", "?") for t in tasks if isinstance(t, dict)]

   802→            item["description"] = data.get("description", "") if isinstance(data, dict) else ""
