   227→# ════════════════════════════════════════════════════════════════ ابزار HTTP و JSON
   228→def _http_json(url: str, headers: Dict[str, str], payload: Dict[str, Any], timeout: int = 180) -> Dict[str, Any]:
   229→    body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
   230→    req = urllib.request.Request(url, data=body, method="POST",
   231→                                 headers={"Content-Type": "application/json",
   232→                                          "User-Agent": "ParsAgent/2.3", **headers})
   233→    opener = _DIRECT if ("127.0.0.1" in url or "localhost" in url) else urllib.request.build_opener()
   234→    try:
   235→        with opener.open(req, timeout=timeout) as r:
   236→            return json.loads(r.read().decode("utf-8"))
   237→    except urllib.error.HTTPError as e:
   238→        try:
   239→            detail = e.read().decode("utf-8", "ignore")[:400]
   240→        except Exception:
   241→            detail = ""
   242→        m = re.search(r'"message"\s*:\s*"([^"]+)"', detail)
   243→        hint = m.group(1) if m else detail[:160]
   244→        raise AgentError(f"خطای {e.code} از سرویس هوش مصنوعی{(' — ' + hint) if hint else ''}")
   245→    except urllib.error.URLError as e:
   246→        raise AgentError(f"ارتباط با سرویس برقرار نشد: {e.reason} — اینترنت/VPN را چک کنید")
   247→    except json.JSONDecodeError:
   248→        raise AgentError("پاسخ سرویس قابل خواندن نبود")

   251→def extract_json(text: str) -> Optional[Any]:
   252→    """اولین JSON متعادلِ داخل متن را بیرون می‌کشد (حتی اگر داخل ``` باشد)."""
   253→    if not text:
   254→        return None
   255→    t = re.sub(r"```(?:json)?", "", text).strip()
   256→    start = t.find("{")
   257→    while start != -1:
   258→        depth, in_str, esc = 0, False, False
   259→        for i in range(start, len(t)):
   260→            ch = t[i]
   261→            if in_str:
   262→                if esc:
   263→                    esc = False
   264→                elif ch == "\\":
   265→                    esc = True
   266→                elif ch == '"':
   267→                    in_str = False
   268→            else:
   269→                if ch == '"':
   270→                    in_str = True
   271→                elif ch == "{":
   272→                    depth += 1
   273→                elif ch == "}":
   274→                    depth -= 1
   275→                    if depth == 0:
   276→                        try:
   277→                            return json.loads(t[start:i + 1])
   278→                        except Exception:
   279→                            break
   280→        start = t.find("{", start + 1)
   281→    return None

   288→# ════════════════════════════════════════════════════════════════ کلاینت LLM
   289→async def llm_chat(ai_cfg: Dict[str, Any], provider: str, model: str,
   290→                   messages: List[Dict[str, str]], *, temperature: float = 0.7,
   291→                   max_tokens: int = 4096, json_mode: bool = False) -> str:
   292→    """گفتگو با هر سرویس — یکسان برای همه (رابط OpenAI/Anthropic/Gemini)."""
   293→    p = PROVIDERS.get(provider)
   294→    if not p:
   295→        raise AgentError(f"سرویس نامعتبر: {provider}")
   296→    if provider == "custom":
   297→        base = (ai_cfg.get("custom_base") or "").strip().rstrip("/")
   298→        model = (ai_cfg.get("custom_model") or "").strip() or model
   299→        if not base:
   300→            raise AgentError("آدرس سرویس سفارشی در تنظیمات خالی است")
   301→    else:
   302→        base = p["base"]
   303→    key = (ai_cfg.get("api_keys") or {}).get(p.get("key_field", provider), "")
   304→    if not key and provider != "custom":
   305→        raise AgentError(f"کلید API «{p['name']}» در تنظیمات وارد نشده — از تب تنظیمات ← هوش مصنوعی")
   306→
   307→    kind = p["kind"]
   308→    if kind == "openai":
   309→        payload: Dict[str, Any] = {"model": model, "messages": messages,
   310→                                    "temperature": temperature, "max_tokens": max_tokens}
   311→        if json_mode:
   312→            payload["response_format"] = {"type": "json_object"}
   313→        headers = {"Authorization": f"Bearer {key}"}
   314→        if provider == "openrouter":
   315→            headers["HTTP-Referer"] = "https://parsagent.local"
   316→            headers["X-Title"] = "ParsAgent"
   317→        url = base + "/chat/completions"
   318→    elif kind == "anthropic":
   319→        system = "\n\n".join(m["content"] for m in messages if m["role"] == "system")
   320→        payload = {"model": model, "max_tokens": max(1024, min(max_tokens, 8000)),
   321→                   "temperature": temperature,
   322→                   "messages": [{"role": m["role"], "content": m["content"]}
   323→                                for m in messages if m["role"] != "system"]}
   324→        if system:
   325→            payload["system"] = system
   326→        headers = {"x-api-key": key, "anthropic-version": "2023-06-01"}
   327→        url = base + "/messages"
   328→    else:  # gemini
   329→        system = "\n\n".join(m["content"] for m in messages if m["role"] == "system")
   330→        payload = {"contents": [{"role": "model" if m["role"] == "assistant" else "user",
   331→                                 "parts": [{"text": m["content"]}]}
   332→                                for m in messages if m["role"] != "system"],
   333→                   "generationConfig": {"temperature": temperature, "maxOutputTokens": max_tokens}}
   334→        if system:
   335→            payload["systemInstruction"] = {"parts": [{"text": system}]}
   336→        if json_mode:
   337→            payload["generationConfig"]["responseMimeType"] = "application/json"
   338→        headers = {"x-goog-api-key": key}
   339→        url = f"{base}/models/{model}:generateContent"
   340→
   341→    last_err: Optional[AgentError] = None
   342→    for attempt in range(2):
   343→        try:
   344→            data = await asyncio.to_thread(_http_json, url, headers, payload)
   345→            if kind == "openai":
   346→                return data["choices"][0]["message"]["content"] or ""
   347→            if kind == "anthropic":
   348→                return "".join(b.get("text", "") for b in data.get("content", []))
   349→            return data["candidates"][0]["content"]["parts"][0]["text"]
   350→        except AgentError as e:
   351→            last_err = e
   352→            if attempt == 0 and "خطای 429" in str(e):
   353→                await asyncio.sleep(3)
   354→                continue
   355→            raise
   356→        except (KeyError, IndexError, TypeError):
   357→            raise AgentError("پاسخ سرویس ساختار مورد انتظار را نداشت (نام مدل درست است؟)")
   358→    raise last_err or AgentError("خطای ناشناخته در سرویس هوش مصنوعی")
   359→

   361→async def llm_json(ai_cfg: Dict[str, Any], provider: str, model: str,
   362→                   messages: List[Dict[str, str]], *, temperature: float = 0.4) -> Dict[str, Any]:
   363→    """گفتگو با خروجی تضمینی JSON — اگر پاسخ خراب بود یک‌بار تلاش مجدد."""
   364→    for attempt in range(2):
   365→        text = await llm_chat(ai_cfg, provider, model, messages,
   366→                              temperature=temperature, json_mode=True)
   367→        data = extract_json(text)
   368→        if isinstance(data, dict):
   369→            return data
   370→        if attempt == 0:
   371→            messages = messages + [
   372→                {"role": "assistant", "content": text[:2000]},
   373→                {"role": "user", "content": "پاسخ قبلی JSON معتبر نبود. فقط و فقط یک JSON درست و کامل بده — بدون هیچ متن اضافه."},
   374→            ]
   375→    raise AgentError("مدل پاسخ JSON درست نداد — دوباره تلاش کنید یا مدل بهتری انتخاب کنید")

   378→# ════════════════════════════════════════════════════════════════ فروشگاه ماموریت‌ها
   379→class MissionStore:
   380→    def __init__(self, path: Path):
   381→        self.path = path
   382→        self.lock = threading.Lock()
   383→        try:
   384→            self.path.parent.mkdir(parents=True, exist_ok=True)
   385→        except Exception:
   386→            pass
   387→
   388→    def load(self) -> List[Dict[str, Any]]:
   389→        try:
   390→            data = json.loads(self.path.read_text(encoding="utf-8"))
   391→            return data if isinstance(data, list) else []
   392→        except Exception:
   393→            return []
   394→
   395→    def save(self, missions: List[Dict[str, Any]]) -> None:
   396→        with self.lock:
   397→            tmp = self.path.with_suffix(".tmp")
   398→            tmp.write_text(json.dumps(missions, ensure_ascii=False, indent=1), encoding="utf-8")
   399→            tmp.replace(self.path)
   400→
   401→    def get(self, mid: str) -> Optional[Dict[str, Any]]:
   402→        return next((m for m in self.load() if m.get("id") == mid), None)
   403→
   404→    def upsert(self, mission: Dict[str, Any]) -> None:
   405→        missions = self.load()
   406→        for i, m in enumerate(missions):
   407→            if m.get("id") == mission["id"]:
   408→                missions[i] = mission
   409→                break
   410→        else:
   411→            missions.insert(0, mission)
   412→        self.save(missions)

   533→    def _launch(self, mid: str) -> None:
   534→        old = self.runners.get(mid)
   535→        if old and not old.done():
   536→            return
   537→        self.runners[mid] = asyncio.create_task(self._runner(mid))

   539→    async def control(self, mid: str, cmd: str) -> Dict[str, Any]:
   540→        m = self.store.get(mid)
   541→        if not m:
   542→            raise AgentError("ماموریت پیدا نشد")
   543→        if cmd == "approve":
   544→            if m["status"] != "wait_approval":
   545→                raise AgentError("این ماموریت منتظر تأیید نیست")
   546→            m["approve_ok"] = True
   547→            self._alog(m, "✅ کاربر برنامه را تأیید کرد — ادامه می‌دهم", "ok")
   548→            self._launch(mid)
   549→        elif cmd == "pause":
   550→            if m["status"] in ("running", "planning"):
   551→                m["pause_flag"] = True
   552→                self._alog(m, "⏸ درخواست توقف ثبت شد — بین گام‌ها متوقف می‌شود", "warn")
   553→        elif cmd == "resume":
   554→            if m["status"] == "paused":
   555→                m["status"] = "running"
   556→                self._alog(m, "▶ ادامهٔ اجرا", "ok")
   557→                self._launch(mid)
   558→            else:
   559→                raise AgentError("این ماموریت قابل ادامه نیست")
   560→        elif cmd == "cancel":
   561→            m["status"] = "cancelled"
   562→            task = self.runners.pop(mid, None)
   563→            if task and not task.done():
   564→                task.cancel()
   565→            self._alog(m, "🛑 ماموریت لغو شد", "warn")
   566→        else:
   567→            raise AgentError("فرمان ناشناخته")
   568→        await self._emit(m)
   569→        return {"ok": True, "status": m["status"]}

   571→    def recover(self) -> None:
   572→        """بعد از ری‌استارت سرور: ماموریت‌های ناتمام → paused تا کاربر ادامه دهد."""
   573→        n = 0
   574→        for m in self.store.load():
   575→            if m.get("status") in ("planning", "running"):
   576→                m["status"] = "paused"
   577→                m["log"].append({"t": datetime.now().strftime("%H:%M:%S"), "level": "warn",
   578→                                 "step": "", "msg": "سرور ری‌استارت شد — با دکمهٔ «ادامه» از همین‌جا ادامه دهید"})
   579→                self.store.upsert(m)
   580→                n += 1
   581→        if n:
   582→            try:
   583→                self.ctx["log"](f"🧠 {n} ماموریت نیمه‌تمام پیدا شد — در حالت توقف", "warn")
   584→            except Exception:
   585→                pass

   587→    async def finalize(self, mid: str) -> None:
   588→        """زمان گزارش رسیده است (زمان‌بند صدایش می‌کند)."""
   589→        m = self.store.get(mid)
   590→        if not m or m["status"] in ("completed", "failed", "cancelled"):
   591→            return
   592→        self._alog(m, "🔔 زمان گزارش ماموریت رسید — گزارش نهایی ساخته می‌شود")
   593→        await self._compile_report(m)
   594→        m["status"] = "completed"
   595→        await self._emit(m)
   596→
   597→    # -------------------------------------------------- اجراکنندهٔ اصلی
   598→    async def _runner(self, mid: str) -> None:
   599→        try:
   600→            while True:
   601→                m = self.store.get(mid)
   602→                if not m or m["status"] in ("completed", "failed", "cancelled", "paused", "waiting"):
   603→                    return
   604→                # ۱) اگر برنامه نیست → بساز
   605→                if not m.get("plan"):
   606→                    ok = await self._do_plan(m)
   607→                    if not ok or m["status"] != "running":
   608→                        return
   609→                steps = m["plan"]["steps"]
   610→                idx = m.get("step_index", 0)
   611→                # ۲) تمام شد؟ → گزارش خودکار
   612→                if idx >= len(steps):
   613→                    if not m.get("report"):
   614→                        await self._compile_report(m)
   615→                    m["status"] = "completed"
   616→                    await self._emit(m)
   617→                    return
   618→                step = steps[idx]
   619→                # ۳) دروازهٔ تأیید
   620→                if m["status"] == "wait_approval":
   621→                    if not m.get("approve_ok"):
   622→                        return
   623→                    m["approve_ok"] = False
   624→                    m["status"] = "running"
   625→                # ۴) انتشار فوری در حالت غیر خودکار نیاز به تأیید دارد
   626→                if step["action"] == "publish_now" and m["autonomy"] != "auto" and not m.get("publish_ok"):
   627→                    m["publish_ok"] = False
   628→                    m["status"] = "wait_approval"
   629→                    self._alog(m, "⏳ انتشار فوری نیازمند تأیید شماست — دکمهٔ «تأیید» را بزنید", "warn", step["action"])
   630→                    await self._emit(m)
   631→                    return
   632→                m["status"] = "running"
   633→                self._alog(m, f"▶ گام {idx + 1} از {len(steps)}: {TOOL_NAMES.get(step['action'], step['action'])} — {step.get('reason') or ''}")
   634→                await self._emit(m)
   635→                try:
   636→                    await self._exec_step(m, idx, step)
   637→                    step["status"] = "done"
   638→                except AgentError as e:
   639→                    step["status"] = "failed"
   640→                    step["error"] = str(e)
   641→                    self._alog(m, f"⚠ این گام ناتمام ماند: {e} — به گام بعدی می‌رویم", "warn", step["action"])
   642→                except asyncio.CancelledError:
   643→                    raise
   644→                except Exception as e:  # noqa
   645→                    step["status"] = "failed"
   646→                    step["error"] = str(e)
   647→                    self._alog(m, f"⚠ خطای غیرمنتظره در گام: {e} — ادامه می‌دهیم", "warn", step["action"])
   648→                m["publish_ok"] = False
   649→                m["step_index"] = idx + 1
   650→                await self._emit(m)
   651→        except asyncio.CancelledError:
   652→            pass
   653→        except Exception as e:  # noqa
   654→            try:
   655→                m = self.store.get(mid)
   656→                if m:
   657→                    m["status"] = "failed"
   658→                    self._alog(m, f"❌ ماموریت متوقف شد: {e}", "error")
   659→                    await self._emit(m)
   660→            except Exception:
   661→                pass

   693→9) report — همین حالا گزارش نهایی می‌سازد. params: {{"include": ["posts","schedules","follows","media","dm"]}}
   694→10) status_check — وضعیت پلتفرم‌ها را بررسی می‌کند. params: {{}}
   695→
   696→نکته‌های مهم:
   697→- اگر ماموریت «ویدیو» یا «ریلز» خواست، از generate_video استفاده کن و بعد خروجی‌اش را برای زمان‌بندی بگذار
   698→- اگر برای پست اینستاگرام/فیسبوک عکس لازم است، به‌جای رهاکردن پست، گام generate_image بگذار
   699→- اگر کاربر گفت «دایرکت‌ها را جواب بده»، dm_autoreply را با persona او تنظیم کن
   700→
   701→قوانین پلتفرم‌ها (برای generate_content):
   702→{rules}
   703→
   704→محدودیت‌های امنی (نقض ممنوع):
   705→- سقف پست روزانه: {max_posts} پست در روز
   706→- سقف فالو روزانه: {caps or '—'}
   707→- اینستاگرام و فیسبوک برای پستِ فید عکس لازم دارند؛ اگر کاربر رسانه نداده، برایشان محتوا تولید کن (پنل موقع زمان‌بندی هشدار می‌دهد)
   708→- برای تلگرام/بله/روبیکا حتماً channels را پر کن؛ اگر کاربر کانالی نگفته، آن پلتفرم را از پست‌ها حذف کن
   709→- گام آخر باید report یا wait_report باشد
   710→
   711→خروجی: فقط یک JSON معتبر — بدون هیچ متن دیگر:
   712→{{"title": "عنوان کوتاه ماموریت", "analysis": "۲ تا ۳ جمله تحلیل و استراتژی کلی", "steps": [{{"action": "...", "params": {{...}}, "reason": "دلیل کوتاه"}}]}}
   713→حداکثر ۸ گام. منطقی و قابل اجرا؛ اعداد را محافظه‌کارانه انتخاب کن."""
   714→        try:
   715→            data = await llm_json(ai, m["provider"], m["model"], [
   716→                {"role": "system", "content": system},
   717→                {"role": "user", "content": "ماموریت من:\n" + m["instruction"]},
   718→            ], temperature=0.3)
   719→        except AgentError as e:
   720→            m["status"] = "failed"
   721→            self._alog(m, f"❌ ساخت برنامه ناموفق: {e}", "error")
   722→            await self._emit(m)
   723→            return False
   724→        steps = []
   725→        for s in (data.get("steps") or [])[:8]:
   726→            act = str(s.get("action") or "").strip()
   727→            if act in TOOL_NAMES and act != "status_check" or act == "status_check":
   728→                steps.append({"action": act, "params": s.get("params") or {},
   729→                              "reason": str(s.get("reason") or "")[:200], "status": "pending"})
   730→        if not steps:
   731→            steps = [{"action": "report", "params": {"include": ["posts", "schedules", "follows"]},
   732→                      "reason": "برنامهٔ معتبری ساخته نشد؛ فقط گزارش", "status": "pending"}]
   733→        m["title"] = str(data.get("title") or "ماموریت " + m["id"][1:6])[:80]
   734→        m["plan"] = {"title": m["title"], "analysis": str(data.get("analysis") or "")[:600], "steps": steps}
   735→        self._alog(m, f"📋 برنامه آماده شد: {len(steps)} گام — {m['title']}", "ok")
   736→        if m["autonomy"] == "manual":
   737→            m["status"] = "wait_approval"
   738→            self._alog(m, "⏸ سطح «پیشنهادی» فعال است — برنامه را ببینید و «تأیید و اجرا» را بزنید", "warn")
   739→            await self._emit(m)
   740→            return False
   741→        m["status"] = "running"
   742→        await self._emit(m)
   743→        return True

   753→    # —— ۱) تولید محتوا
   754→    async def _tool_generate_content(self, m: Dict[str, Any], p: Dict[str, Any]) -> None:

   766→                          f"{PLATFORM_INFO[pid]['hashtags']} هشتگ (ته‌متن، بعد از یک خط خالی) · سبک: {PLATFORM_INFO[pid]['style']}"

   769→        note = p.get("note") or ""
   770→        n_posts = 0
   771→        for idea in ideas:
   772→            if len(m["posts"]) >= cap_total:
   773→                self._alog(m, f"⏹ سقف {cap_total} پست برای هر ماموریت پر شد — بقیهٔ ایده‌ها رد شد", "warn", "generate_content")
   774→                break
   775→            self._alog(m, f"✍️ تولید محتوا برای ایده: «{idea[:60]}»")
   776→            await self._emit(m)
   777→            data = await llm_json(ai, m["provider"], m["model"], [
   778→                {"role": "system", "content":
   779→                 f"تو نویسندهٔ حرفه‌ای شبکه‌های اجتماعی فارسی‌زبان هستی. برای این ایده، برای هر پلتفرمِ خواسته‌شده دقیقاً یک پست بنویس.\n\n"
   780→                 f"قوانین پلتفرم‌ها:\n{rules}\n\nلحن: {tone}\n{('نکتهٔ کاربر: ' + note) if note else ''}\n\n"
   781→                 "متن‌ها کاملاً فارسی و اصیل باشند؛ کلیشه ننویس. best_time = ساعت پیشنهادی انتشار برای مخاطب ایرانی (HH:MM).\n"
   782→                 "برای هر پست یک image_prompt هم بنویس: توضیح انگلیسیِ کوتاه عکسِ پیشنهادی همان پست (برای تصویرسازی با AI).\n"
   783→                 'خروجی فقط JSON: {"posts": [{"platform": "instagram", "text": "...", "hashtags": "#a #b", "best_time": "18:30", "image_prompt": "cinematic photo of ..."}]}'},
   784→                {"role": "user", "content": f"ایده: {idea}\nپلتفرم‌ها: {', '.join(platforms)}"},
   785→            ], temperature=0.8)
   786→            got = 0
   787→            for raw in (data.get("posts") or [])[:per_idea * len(platforms)]:
   788→                pid = raw.get("platform")
   789→                if pid not in platforms or len(m["posts"]) >= cap_total:
   790→                    continue
   791→                info = PLATFORM_INFO[pid]

   793→                if not text:
   794→                    continue
   795→                m["posts"].append({
   796→                    "id": f"p{len(m['posts']) + 1}", "idea": idea[:120], "platform": pid,
   797→                    "text": text, "hashtags": str(raw.get("hashtags") or "").strip()[:200],
   798→                    "best_time": str(raw.get("best_time") or "")[:5], "status": "draft",
   799→                    "image_prompt": str(raw.get("image_prompt") or "").strip()[:400],
   800→                    "media": "",
   801→                })
   802→                got += 1
   803→            n_posts += got
   804→            self._alog(m, f"✅ {got} پست برای «{idea[:40]}» تولید شد (مجموع: {len(m['posts'])})", "ok", "generate_content")
   805→            await self._emit(m)
   806→        if not m["posts"]:
   807→            raise AgentError("هیچ پستی تولید نشد — ایده یا پلتفرم‌ها را واضح‌تر بنویسید")

   863→    async def _tool_schedule_publish(self, m: Dict[str, Any], p: Dict[str, Any]) -> None:
   864→        ai = self._ai_cfg()
   865→        strategy = p.get("strategy") or "spread_week"
   866→        per_day = max(1, min(6, int(p.get("per_day") or ai.get("budgets", {}).get("posts_per_day", 3) or 3)))
   867→        channels = p.get("channels") or {}
   868→        images = list(self.ctx["media_images"]())
   869→        posts = [x for x in m["posts"] if x.get("status") == "draft"]
   870→        if not posts:
   871→            self._alog(m, "پست آماده‌ای برای زمان‌بندی نیست (اول generate_content)", "warn", "schedule_publish")
   872→            return
   873→        # پخش زمان‌ها: از امروز، حداکثر per_day پست در روز، در ساعات پیشنهادی
   874→        times: List[datetime] = []
   875→        now = datetime.now()
   876→        day_offset = 0
   877→        hour_pool = list(DEFAULT_HOURS)
   878→        for i in range(len(posts)):
   879→            d = now + timedelta(days=day_offset)
   880→            if strategy == "best_times":
   881→                bt = (posts[i].get("best_time") or "").strip()
   882→                try:
   883→                    hh, mm = map(int, bt.split(":")[:2])
   884→                except Exception:
   885→                    hh, mm = hour_pool[i % len(hour_pool)], random.randint(2, 40)
   886→            else:
   887→                hh, mm = hour_pool[i % len(hour_pool)], random.randint(2, 40)
   888→            if hh < 8 or hh >= 24:
   889→                hh = 21
   890→            when = d.replace(hour=hh, minute=mm, second=0, microsecond=0)
   891→            if when <= now:
   892→                when += timedelta(days=1)
   893→            times.append(when)
   894→            if (i + 1) % per_day == 0:
   895→                day_offset += 1
   896→        if strategy == "same_time":
   897→            base_t = times[0]
   898→            times = [base_t + timedelta(minutes=7 * i) for i in range(len(times))]
   899→        n_ok, n_skip = 0, 0
   900→        for post, when in zip(posts, times):
   901→            pid = post["platform"]
   902→            info = PLATFORM_INFO[pid]
   903→            params: Dict[str, Any] = {"text": post["text"] + (
   904→                "\n\n" + post["hashtags"] if post.get("hashtags") else "")}
   905→            if info["is_messenger"]:
   906→                target = (channels.get(pid) or "").strip()
   907→                if not target:
   908→                    n_skip += 1
   909→                    post["status"] = "no_channel"
   910→                    continue
   911→                params["target"] = target
   912→            elif info["needs_media"]:
   913→                # ۲.۴.۰: اگر کتابخانه عکس ندارد، ایجنت خودش با AI عکس می‌سازد
   914→                media = post.get("media") or ""
   915→                if not media:
   916→                    images = list(self.ctx["media_images"]())
   917→                    media = images[0] if images else ""
   918→                if media:
   919→                    params["media"] = media
   920→                elif callable(self.ctx.get("generate_media_images")):
   921→                    try:
   922→                        self._alog(m, f"🎨 عکس پست {post['id']} با AI ساخته می‌شود…", "", "generate_image")
   923→                        await self._emit(m)
   924→                        prompt = post.get("image_prompt") or (post["idea"] + " — high quality")
   925→                        res = await self.ctx["generate_media_images"]([prompt], 1, "square", m.get("id"))
   926→                        files = (res or {}).get("files") or []
   927→                        if files:
   928→                            params["media"] = files[0]
   929→                            post["media"] = files[0]
   930→                        else:
   931→                            n_skip += 1
   932→                            post["status"] = "needs_media"
   933→                            continue
   934→                    except Exception as e:  # noqa
   935→                        self._alog(m, f"⚠ ساخت عکس ناموفق: {e}", "warn", "generate_image")
   936→                        n_skip += 1
   937→                        post["status"] = "needs_media"
   938→                        continue
   939→                else:
   940→                    n_skip += 1
   941→                    post["status"] = "needs_media"
   942→                    continue
   943→            entry = {
   944→                "title": f"AI · پست {post['id']} — {info['name']}",
   945→                "type": "action", "platform": pid, "action": info["post_action"],
   946→                "params": params,
   947→                "timing": {"type": "once", "at": when.isoformat(timespec="seconds")},
   948→                "pacing": {}, "enabled": True, "mission_id": m["id"], "post_id": post["id"],
   949→            }
   950→            job = await self.ctx["add_schedule"](entry)
   951→            post["status"] = "scheduled"
   952→            m["scheduled"].append({"post_id": post["id"], "platform": pid,
   953→                                   "at": when.isoformat(timespec="seconds"),
   954→                                   "job_id": job.get("id"), "action": info["post_action"]})
   955→            n_ok += 1
   956→        msg = f"📅 {n_ok} پست در صف انتشار قرار گرفت" + (
   957→            f" — {n_skip} مورد رد شد (عکس یا کانال ندارند؛ وضعیت‌شان در لیست پست‌ها مشخص است)" if n_skip else "")
   958→        self._alog(m, msg, "ok" if n_ok else "warn", "schedule_publish")
   959→        if n_skip:
   960→            self._alog(m, "💡 برای پست‌های اینستاگرام/فیسبوک از تب «رسانه‌ها» عکس بگذارید؛ برای پیام‌رسان‌ها نام کانال را در ماموریت بگویید", "warn")

  1000→    # —— ۴) استراتژی فالو
  1001→    async def _tool_follow_strategy(self, m: Dict[str, Any], p: Dict[str, Any]) -> None:
  1002→        pid = p.get("platform")
  1003→        if pid not in FOLLOW_CAPS or pid not in PLATFORM_INFO:
  1004→            raise AgentError(f"برای «{pid}» استراتژی فالو پشتیبانی نمی‌شود "
  1005→                             f"(فقط: {', '.join(FOLLOW_CAPS)})")
  1006→        cap = FOLLOW_CAPS[pid]
  1007→        daily = max(1, min(cap["cap"], int(p.get("daily_follows") or 10)))
  1008→        source = (p.get("source") or "").strip()
  1009→        if not source:
  1010→            raise AgentError("منبع فالو مشخص نیست — هشتگ یا لیست کاربران لازم است")
  1011→        hours = [int(h) for h in (p.get("active_hours") or []) if 6 <= int(h) <= 23] or [10, 14, 21]
  1012→        hours = sorted(set(hours))[:4]
  1013→        per_run = max(1, daily // len(hours))
  1014→        plan = {"daily_follows": daily, "per_run": per_run, "hours": hours,
  1015→                "source": source, "created_at": _now_iso(), "mission_id": m["id"]}
  1016→        # ثبت در تنظیمات برای نمایش در پنل
  1017→        try:
  1018→            st = self.ctx["get_settings"]()
  1019→            st.setdefault("ai", {}).setdefault("follow_plans", {})[pid] = plan
  1020→            self.ctx["save_settings"](st)
  1021→        except Exception:
  1022→            pass
  1023→        # ساخت زمان‌بندی‌های روزانه
  1024→        for h in hours:
  1025→            entry = {
  1026→                "title": f"AI · فالو {PLATFORM_INFO[pid]['name']} ساعت {h}",
  1027→                "type": "action", "platform": pid, "action": cap["action"],
  1028→                "params": {"target": source, "count": per_run},
  1029→                "timing": {"type": "daily", "hours": [h], "minute": random.randint(2, 25)},
  1030→                "pacing": {"delay_min": 45, "delay_max": 140},
  1031→                "enabled": True, "mission_id": m["id"],
  1032→            }
  1033→            await self.ctx["add_schedule"](entry)
  1034→        self._alog(m, f"🎯 استراتژی فالو ثبت شد: {daily} نفر در روز ({per_run} نفر در {len(hours)} نوبت ساعت‌های "
  1035→                     f"{'، '.join(str(h) for h in hours)}) از «{source}» — سقف امن {cap['cap']}", "ok", "follow_strategy")
