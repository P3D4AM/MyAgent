   172→   172→  /** Advances the simulation. `dt` is capped so long tab sleeps don't jump. */
   173→   173→  tick(): void {
   174→   174→    const t = now();
   175→   175→    let dt = t - this.lastTick;
   176→   176→    this.lastTick = t;
   177→   177→    if (dt <= 0) return;
   178→   178→    if (dt > 6000) dt = 6000;
   179→   179→    const dtS = dt / 1000;
   180→   180→
   181→   181→    this.system.uptimeS += dtS;
   182→   182→
   183→   183→    // ---- vitals random-walk (busier when tasks are running) ----
   184→   184→    const activeTasks = this.tasks.filter((task) => task.status === "running").length;
   185→   185→    const load = 6 + activeTasks * 22 + (this.recorderActive ? 6 : 0);
   186→   186→    this.system.cpu = clamp(this.system.cpu + rand(-4, 4) + (load - this.system.cpu) * 0.08, 2, 97);
   187→   187→    this.system.ram = clamp(this.system.ram + rand(-1.5, 1.6), 24, 88);
   188→   188→
   189→   189→    // ---- login flows ----
   190→   190→    for (const p of this.platforms.values()) {
   191→   191→      if (!p.loggingIn) {
   192→   192→        if (p.loggedIn) p.sessionAgeMin += dtS / 60;
   193→   193→        continue;
   194→   194→      }
   195→   195→      const elapsed = t - (p.loginStartedAt ?? t);
   196→   196→      const dur = p.loginDurationMs;
   197→   197→      // staged progress logs at 35% and 70%
   198→   198→      if (elapsed >= dur * 0.35 && elapsed - dt < dur * 0.35) {
   199→   199→        this.log(p.id, "info", "waiting for your login in the opened window…");
   200→   200→      }
   201→   201→      if (elapsed >= dur * 0.7 && elapsed - dt < dur * 0.7) {
   202→   202→        this.log("playwright", "info", "storage state captured (localStorage + cookies)");
   203→   203→      }
   204→   204→      if (elapsed >= dur) {
   205→   205→        p.loggingIn = false;
   206→   206→        p.loggedIn = true;
   207→   207→        p.running = true;
   208→   208→        p.channel = p.channel ?? this.settings.channelFallbacks[0];
   209→   209→        p.sessionAgeMin = 0;
   210→   210→        p.lastActiveAt = t;
   211→   211→        this.log(p.id, "ok", `login complete — session saved → data/sessions/${p.id}.json`);
   212→   212→        this.log("playwright", "ok", "persistent profile will skip login next time");
   213→   213→      }
   214→   214→    }
   215→   215→
   216→   216→    // ---- tasks ----
   217→   217→    for (const task of this.tasks) {
   218→   218→      if (task.status === "running") this.advanceTask(task, dt, t);
   219→   219→      // scheduler
   220→   220→      if (
   221→   221→        task.status !== "running" &&
   222→   222→        task.schedule.type !== "manual" &&
   223→   223→        task.nextRunAt !== null &&
   224→   224→        t >= task.nextRunAt
   225→   225→      ) {
   226→   226→        this.log("scheduler", "info", `schedule triggered → ${task.name}`);
   227→   227→        this.startTask(task, t);
   228→   228→      }
   229→   229→    }
   230→   230→
   231→   231→    // ---- recorder ----
   232→   232→    if (this.recorderActive) this.generateRecordEvents(dt, t);
   233→   233→
   234→   234→    // ---- playback ----
   235→   235→    if (this.playback) {
   236→   236→      const pb = this.playback;
   237→   237→      const elapsed = t - pb.startedAt;
   238→   238→      const frac = clamp(elapsed / pb.durationMs, 0, 1);
   239→   239→      // move the simulated mouse along the macro path
   240→   240→      const macro = this.macros.find((m) => m.id === pb.macroId);
   241→   241→      if (macro) {
   242→   242→        const idx = Math.min(
   243→   243→          macro.events.length - 1,
   244→   244→          Math.floor(frac * macro.events.length),
   245→   245→        );
   246→   246→        const ev = macro.events[idx];
   247→   247→        if (ev && ev.x !== undefined && ev.y !== undefined) {
   248→   248→          this.system.mouse = { x: ev.x, y: ev.y };
   249→   249→        }
   250→   250→      }
   251→   251→      const quarter = Math.floor(frac * 4);
   252→   252→      if (quarter > this.playbackLogged && quarter < 4) {
   253→   253→        this.playbackLogged = quarter;
   254→   254→        this.log(
   255→   255→          "desktop",
   256→   256→          "info",
   257→   257→          `replaying ${macro?.name ?? "macro"} — event ${Math.floor(frac * (pb.totalEvents || 1))}/${pb.totalEvents}`,
   258→   258→        );
   259→   259→      }
   260→   260→      if (elapsed >= pb.durationMs) {
   261→   261→        const macroName = pb.macroName;
   262→   262→        this.playback = null;
   263→   263→        this.playbackLogged = 0;
   264→   264→        this.log("desktop", "ok", `macro replayed — ${macroName} (${pb.totalEvents} events at ×${pb.speed})`);
   265→   265→      }
   266→   266→    }
   267→   267→  }

   315→   315→  private buildSteps(
   316→   316→    platformId: string,
   317→   317→    actionId: string,
   318→   318→    engine: EngineId,
   319→   319→    params: { text: string; target: string; image: string },
   320→   320→  ): { steps: TaskStepState[]; totalWeight: number } {
   321→   321→    const platform = getPlatform(platformId);
   322→   322→    const action = platform.actions.find((a) => a.id === actionId) ?? platform.actions[0];
   323→   323→    const script = buildScript(platform, action, params, {
   324→   324→      engine,
   325→   325→      networkIssue: false,
   326→   326→    });
   327→   327→    const totalWeight = script.reduce((acc, s) => acc + s.weight, 0);
   328→   328→    let acc = 0;
   329→   329→    const steps: TaskStepState[] = script.map((s) => {
   330→   330→      acc += s.weight;
   331→   331→      return {
   332→   332→        tag: s.tag,
   333→   333→        level: s.level,
   334→   334→        text: s.text,
   335→   335→        at: acc / totalWeight,
   336→   336→        emitted: false,
   337→   337→      };
   338→   338→    });
   339→   339→    // emit the very first step instantly
   340→   340→    if (steps.length > 0) steps[0].emitted = true;
   341→   341→    return { steps, totalWeight };
   342→   342→  }

   355→   355→    if (p) {
   356→   356→      p.lastAction = task.actionLabel;

   359→   359→        p.running = true;

   361→   361→      }
   362→   362→    }
   363→   363→    this.log("gate", "info", `run → ${task.name} (engine=${task.engine}  speed=${SPEEDS[task.speed]?.label})`);

   369→   369→  private advanceTask(task: PanelTask, dt: number, t: number): void {
   370→   370→    const prev = task.progress;
   371→   371→    task.progress = clamp(task.progress + dt / task.simDurationMs, 0, 1);
   372→   372→    // emit crossed steps
   373→   373→    for (let i = 0; i < task.steps.length; i++) {
   374→   374→      const s = task.steps[i];
   375→   375→      if (!s.emitted && task.progress >= s.at * 0.999) {
   376→   376→        s.emitted = true;
   377→   377→        task.currentStep = i + 1;
   378→   378→        this.log(s.tag, s.level, s.text);
   379→   379→      }
   380→   380→    }
   381→   381→    if (task.progress >= 1 && prev < 1) {
   382→   382→      task.status = "done";
   383→   383→      task.lastDurationS = (t - (task.startedAt ?? t)) / 1000;
   384→   384→      task.progress = 1;
   385→   385→      this.scheduleNext(task, t);
   386→   386→    }
   387→   387→  }

   393→   393→    } else {
   394→   394→      task.nextRunAt = null;
   395→   395→    }
   396→   396→  }
