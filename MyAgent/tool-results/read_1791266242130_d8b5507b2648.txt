   116→def _findable(name: str) -> bool:
   117→    """آیا ماژول قابل ایمپورت است؟ (مقاوم به ماژول‌های بی-spec)"""

   119→        return importlib.util.find_spec(name) is not None
   120→    except Exception:
   121→        return name in sys.modules

   128→def _human(n) -> str:
   129→    x = float(n)
   130→    for unit in ("B", "KB", "MB", "GB", "TB"):
   131→        if x < 1024:
   132→            return f"{x:.1f} {unit}"
   133→        x /= 1024
   134→    return f"{x:.1f} PB"

   147→def _is_free_threaded() -> bool:

   149→    سازگار نیست — همان خطای عجیب «کامپایل Rust» که دیدید."""

   154→        return bool(int(sysconfig.get_config_var("Py_GIL_DISABLED") or 0))

   156→        return False

   184→def compat_problems(info) -> list:
   185→    """لیست ناسازگاری‌های سیستم فعلی با بستهٔ ساخته‌شده (خالی = مشکلی نیست)."""
   186→    probs = []
   187→    if _is_free_threaded():
   188→        probs.append("پایتون این سیستم از نوع free-threaded است (شناسه‌ای مثل cp313t). "
   189→                     "wheelهای این بسته با آن سازگار نیستند؛ نسخهٔ «استاندارد» همان پایتون را "
   190→                     "از python.org نصب کنید (تیک Add to PATH).")
   191→    if info:
   192→        if info.get("platform") and info["platform"] != platform.system():
   193→            probs.append(f"بسته برای {info['platform']} ساخته شده ولی این سیستم "
   194→                        f"{platform.system()} است.")
   195→        mine = _python_tag()
   196→        if info.get("python_tag") and mine != info["python_tag"]:
   197→            probs.append(f"پایتون این سیستم {mine} است ولی بسته برای "
   198→                        f"{info['python_tag']} ساخته شده — همان نسخهٔ پایتون را نصب کنید.")
   199→        bits = 64 if platform.architecture()[0].startswith("64") else 32
   200→        if info.get("bitness") and bits != info["bitness"]:
   201→            probs.append(f"پایتون این سیستم {bits} بیتی است ولی بسته {info['bitness']} بیتی "
   202→                         "ساخته شده است.")
   203→    return probs

   206→def disk_msg() -> str | None:
   207→    """اگر فضای دیسک برای نصبِ بسته کافی نباشد، پیام فارسی برمی‌گرداند."""
   208→    if not BUNDLE_DIR.is_dir():
   209→        return None
   210→    need = int(_dir_size(BUNDLE_DIR) * 1.15) + 50 * 1024 * 1024
   211→    free = shutil.disk_usage(BASE).free
   212→    if free < need:
   213→        return (f"فضای خالی درایو کافی نیست — حجم بسته {_human(_dir_size(BUNDLE_DIR))} "
   214→                f"است ولی فقط {_human(free)} آزاد دارید (خطای os error 112). "
   215→                "اول فضا آزاد کنید.")
   216→    return None

   219→def _run_stream(cmd, log=_console_log, env=None) -> int:
   220→    """اجرای فرمان با استریم زندهٔ خروجی در لاگ؛ کد خروج را برمی‌گرداند."""
   221→    try:
   222→        p = subprocess.Popen(
   223→            [str(c) for c in cmd], cwd=str(BASE),
   224→            stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
   225→            text=True, encoding="utf-8", errors="replace",
   226→            env=env or os.environ, creationflags=NO_WINDOW)
   227→    except Exception as exc:
   228→        log("err", f"اجرای فرمان ناموفق بود: {exc}")
   229→        return 1
   230→    for line in p.stdout:
   231→        log("pip", line.rstrip())
   232→    return p.wait()

   240→def build_offline_bundle(include_browsers: bool = True, log=_console_log) -> str:
   241→    """ساخت بستهٔ آفلاین داخل پروژه. خلاصه را برمی‌گرداند؛ در خطا RuntimeError."""
   242→    req = BASE / "requirements.txt"
   243→    if not req.is_file():
   244→        raise RuntimeError("فایل requirements.txt کنار launcher.py پیدا نشد.")
   245→
   246→    need_gb = 2.5 if include_browsers else 1.2
   247→    free_gb = shutil.disk_usage(BASE).free / 1024 ** 3
   248→    if free_gb < need_gb:
   249→        raise RuntimeError(f"فضای خالی کافی نیست — حداقل {need_gb:.1f} گیگابایت لازم است "
   250→                           f"ولی الان {free_gb:.1f} گیگابایت آزاد است (خطای os error 112).")
   251→
   252→    BUNDLE_DIR.mkdir(parents=True, exist_ok=True)
   253→    WHEELS_DIR.mkdir(parents=True, exist_ok=True)
   254→    shutil.copy2(req, BUNDLE_DIR / "requirements.txt")
   255→
   256→    # --- ۱) دانلود همهٔ پیش‌نیازها به‌صورت wheel آماده ---------------
   257→    log("info", "۱) دانلود پیش‌نیازها به‌صورت wheel آماده (چند دقیقه)…")
   258→    base = [sys.executable, "-m", "pip", "download", "-r", str(req),
   259→            "-d", str(WHEELS_DIR), "--progress-bar", "off"]
   260→    rc = _run_stream(base + ["--only-binary=:all:"], log=log)
   261→    if rc != 0:
   262→        log("warn", "دانلود با محدودیت wheel-only نشد؛ تلاش مجدد بدون محدودیت…")
   263→        rc = _run_stream(base, log=log)
   264→        if rc != 0:
   265→            raise RuntimeError("دانلود پکیج‌ها از اینترنت ناموفق بود.")
   266→        log("warn", "بعضی پکیج‌ها سورس‌اند و ممکن است روی سیستم مقصد کامپایلر بخواهند.")
   267→    # خود pip هم داخل بسته باشد
   268→    _run_stream([sys.executable, "-m", "pip", "download", "pip",
   269→                 "-d", str(WHEELS_DIR), "--only-binary=:all:",
   270→                 "--progress-bar", "off"], log=log)
   271→
   272→    # --- ۲) مرورگر کرومیوم (اختیاری) --------------------------------
   273→    if include_browsers:
   274→        log("info", "۲) آماده‌سازی مرورگر کرومیوم…")
   275→        if not _findable("playwright"):
   276→            log("info", "نصب موقت playwright از روی wheelهای تازه دانلودشده…")
   277→            rc = _run_stream([sys.executable, "-m", "pip", "install", "--no-index",
   278→                              "--find-links", str(WHEELS_DIR), "playwright",
   279→                              "--progress-bar", "off"], log=log)
   280→            if rc != 0:
   281→                raise RuntimeError("نصب playwright از روی wheelها ناموفق بود.")
   282→        if not _has_chromium():
   283→            log("warn", "کرومیوم نصب نیست؛ دانلود از آینهٔ npmmirror (بدون فیلترشکن)…")
   284→            env = {**os.environ, "PLAYWRIGHT_DOWNLOAD_HOST": PW_MIRROR}
   285→            rc = _run_stream([sys.executable, "-m", "playwright", "install", "chromium"],
   286→                             log=log, env=env)
   287→            if rc != 0:
   288→                raise RuntimeError("دانلود کرومیوم ناموفق بود.")
   289→        src = _browsers_cache()
   290→        if not src.is_dir():
   291→            raise RuntimeError(f"پوشهٔ مرورگرهای Playwright پیدا نشد: {src}")
   292→        dst = BUNDLE_DIR / "ms-playwright"
   293→        dst.mkdir(parents=True, exist_ok=True)
   294→        kept = [d.name for d in src.iterdir()
   295→                if d.is_dir() and (d.name.startswith("chromium") or d.name.startswith("ffmpeg"))]
   296→        if not kept:
   297→            raise RuntimeError("در کش Playwright هیچ مرورگری برای کپی پیدا نشد.")
   298→        for name in kept:
   299→            shutil.copytree(src / name, dst / name, dirs_exist_ok=True)
   300→        log("ok", "مرورگرها داخل بسته کپی شدند: " + "، ".join(kept))
   301→    else:
   302→        log("warn", "مرورگر داخل بسته نشد (حالت سبک) — نصب آفلاین بعداً از "
   303→                   "Edge/Chrome خود ویندوز استفاده می‌کند.")
   304→
   305→    # --- ۳) شناسنامهٔ بسته ------------------------------------------
   306→    info = {
   307→        "platform": platform.system(),
   308→        "python_version": platform.python_version(),
   309→        "python_tag": _python_tag(),
   310→        "soabi": sysconfig.get_config_var("SOABI"),
   311→        "bitness": 64 if platform.architecture()[0].startswith("64") else 32,
   312→        "with_browsers": include_browsers,
   313→        "created": time.strftime("%Y-%m-%d %H:%M"),
   314→        "app_version": "2.7.0",
   315→    }
   316→    (BUNDLE_DIR / "bundle_info.json").write_text(
   317→        json.dumps(info, indent=2, ensure_ascii=False), encoding="utf-8")
   318→
   319→    n_whl = len(list(WHEELS_DIR.glob("*.whl")))
   320→    w = _dir_size(WHEELS_DIR)
   321→    b = _dir_size(BUNDLE_DIR / "ms-playwright")
   322→    return (f"تعداد wheelها: {n_whl}\n"
   323→            f"حجم wheelها: {_human(w)}\n"
   324→            + (f"حجم مرورگرها: {_human(b)}\n" if b else "")
   325→            + f"حجم کل بسته: {_human(w + b)}\n"
   326→            f"محل بسته: {BUNDLE_DIR}")

   329→def install_from_bundle(log=_console_log):
   330→    """نصب کامل از بستهٔ آفلاین — بدون حتی یک بایت اینترنت.
   331→    برمی‌گرداند: (موفق؟، لیست ماژول‌های همچنان نصب‌نشده)"""
   332→    log("info", "نصب آفلاین از پوشهٔ offline_bundle (بدون اینترنت)…")
   333→    rc = _run_stream([sys.executable, "-m", "pip", "install",
   334→                      "--no-index", "--find-links", str(WHEELS_DIR),
   335→                      "-r", str(BUNDLE_DIR / "requirements.txt"),
   336→                      "--progress-bar", "off"], log=log)
   337→    if rc != 0:
   338→        log("err", "نصب پکیج‌ها از بسته ناموفق بود (لاگ بالا را ببینید).")
   339→        return False, check_missing()
   340→
   341→    src = BUNDLE_DIR / "ms-playwright"
   342→    if src.is_dir():
   343→        dst = _browsers_cache()
   344→        dst.mkdir(parents=True, exist_ok=True)
   345→        for item in src.iterdir():
   346→            if item.is_dir():
   347→                shutil.copytree(item, dst / item.name, dirs_exist_ok=True)
   348→            else:
   349→                shutil.copy2(item, dst / item.name)
   350→        log("ok", f"مرورگرها در محل Playwright کپی شدند: {dst}")
   351→    else:
   352→        log("warn", "بسته شامل مرورگر نیست — از Edge/Chrome خود ویندوز استفاده می‌شود.")
   353→
   354→    missing = check_missing()
   355→    for m in REQUIRED:
   356→        log("ok" if m not in missing else "err",

   625→    def _run_async(self, cmd, env=None, on_done=None, cwd=None):
   626→        def work():
   627→            try:
   628→                p = subprocess.Popen(
   629→                    cmd, cwd=str(cwd or BASE),
   630→                    stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
   631→                    text=True, encoding="utf-8", errors="replace",
   632→                    env=env or os.environ, creationflags=NO_WINDOW)
   633→                for line in p.stdout:
   634→                    self.post(lambda l=line: self.log("pip", l.rstrip()))
   635→                code = p.wait()
   636→                self.post(lambda: on_done(code == 0) if on_done else None)
   637→            except Exception as exc:
   638→                self.post(lambda: self.log("err", f"خطا: {exc}"))
   639→                self.post(lambda: on_done(False) if on_done else None)
   640→        threading.Thread(target=work, daemon=True).start()

   651→        self.log("info", "شروع نصب پیش‌نیازها (ممکن است چند دقیقه طول بکشد)…")
   652→        env = {**os.environ, "PYTHONIOENCODING": "utf-8"}
   653→        self._run_async(
   654→            [sys.executable, "-m", "pip", "install", "-r", "requirements.txt"],
   655→            env=env, on_done=self._after_install)

   695→    def build_offline(self):

   697→            messagebox.showinfo("مشغول", "یک نصب/ساخت دیگر در جریان است")
   698→            return
   699→        if self.proc and self.proc.poll() is None:
   700→            messagebox.showwarning("در حال اجرا",
   701→                                   "اول سرور را متوقف کن، بعد بسته را بساز")
   702→            return

   704→        if not messagebox.askyesno(

   711→            return
   712→        self.installing = True
   713→        self.set_status("در حال ساخت بستهٔ آفلاین…", COLORS["sky"])
   714→        self.log("info", "شروع ساخت بستهٔ آفلاین…")

   716→        def work():
   717→            try:
   718→                summary = build_offline_bundle(include, self._lg)
   719→                self.post(lambda: self._bundle_done(summary))
   720→            except Exception as exc:
   721→                self.post(lambda e=exc: self._offline_failed("ساخت بسته", e))
   722→        threading.Thread(target=work, daemon=True).start()

   737→        self.log("err", f"{what} ناموفق: {exc}")
   738→        messagebox.showerror(what, str(exc))
   739→
   740→    def install_offline_clicked(self, auto_start: bool = False):
   741→        if self.installing:
   742→            messagebox.showinfo("مشغول", "یک نصب/ساخت دیگر در جریان است")
   743→            return
   744→        if self.proc and self.proc.poll() is None:
   745→            messagebox.showwarning("در حال اجرا",
   746→                                   "اول سرور را متوقف کن، بعد نصب را بزن")
   747→            return
   748→        if not bundle_available():
   749→            messagebox.showerror(
   750→                "بسته پیدا نشد",
   751→                "پوشهٔ offline_bundle کنار پروژه نیست.\n"
   752→                "اول روی یک سیستم اینترنت‌دار با «💾 ساخت بستهٔ آفلاین» "
   753→                "آن را بساز و کل پوشهٔ پروژه را انتقال بده.")
   754→            return
   755→        probs = compat_problems(read_bundle_info())
   756→        if probs and not messagebox.askyesno(
   757→                "⚠ ناسازگاری بسته",
   758→                "با این بستهٔ آفلاین ناسازگاری وجود دارد:\n\n"
   759→                + "\n".join(probs)
   760→                + "\n\nبه‌هرحال ادامه بدهی؟"):
   761→            self.log("warn", "نصب آفلاین به دلیل ناسازگاری لغو شد")
   762→            return
   763→        msg = disk_msg()
   764→        if msg:
   765→            messagebox.showerror("فضای دیسک", msg)

   767→        self.installing = True
   768→        self.set_status("در حال نصب آفلاین (بدون اینترنت)…", COLORS["sky"])
   769→        self.log("info", "شروع نصب آفلاین…")
   770→
   771→        def work():
   772→            try:
   773→                ok, missing = install_from_bundle(self._lg)
   774→                self.post(lambda: self._offinstall_done(ok, missing, auto_start))
   775→            except Exception as exc:
   776→                self.post(lambda e=exc: self._offline_failed("نصب آفلاین", e))
   777→        threading.Thread(target=work, daemon=True).start()
   778→
   779→    def _offinstall_done(self, ok: bool, missing: list, auto_start: bool):
   780→        self.installing = False
   781→        if ok:
   782→            self.set_running(False)
   783→            self.log("ok", "نصب آفلاین کامل شد ✓")
   784→            if auto_start or messagebox.askyesno(
   785→                    "نصب کامل ✓",
   786→                    "همهٔ پیش‌نیازها بدون اینترنت نصب شد ✓\n\nسرور الان اجرا شود؟"):
   787→                self.start_server()
   788→        else:
   789→            self.set_status("نصب ناقص", COLORS["err"])
   790→            self.log("err", "این ماژول‌ها هنوز نصب نیستند: " + "، ".join(missing))
   791→            messagebox.showerror("نصب ناقص",
   792→                                 "این ماژول‌ها نصب نشدند:\n" + "، ".join(missing))

   797→    def start_server(self):
   798→        if self.installing:
   799→            messagebox.showinfo("در حال نصب",
   800→                                "صبر کن نصب پیش‌نیازها تمام شود")
   801→            return
   802→        if self.proc and self.proc.poll() is None:
   803→            self.log("info", "سرور از قبل در حال اجراست — پنل باز می‌شود")
   804→            webbrowser.open(PANEL_URL)
   805→            return
   806→        # اگر سرورِ دیگری (مثلاً از اجرای قبلی) روی پورت روشن است:
   807→        if server_up(0.8):
   808→            self.log("ok", "یک نسخه از سرور روی این پورت در حال اجراست — پنل باز می‌شود")
   809→            self.server_ready = True
   810→            self.set_running(True)
   811→            webbrowser.open(PANEL_URL)
   812→            return
   813→        if check_missing():
   814→            self._ask_install_smart(check_missing())
   815→            return
   816→
   817→        self.server_ready = False
   818→        self.log("info", f"راه‌اندازی سرور پنل روی {PANEL_URL} …")
   819→        self.set_status("در حال راه‌اندازی سرور…", COLORS["sky"])
   820→        env = {**os.environ, "PYTHONIOENCODING": "utf-8",
   821→               "NO_COLOR": "1",
   822→               "NO_PROXY": "127.0.0.1,localhost",
   823→               "no_proxy": "127.0.0.1,localhost"}
   824→        try:
   825→            # خروجی سرور گرفته می‌شود تا هر خطری در لاگ لانچر دیده شود
   826→            self.proc = subprocess.Popen(
   827→                [sys.executable, "-u", str(BASE / "panel_server.py")],
   828→                cwd=str(BASE), stdout=subprocess.PIPE,
   829→                stderr=subprocess.STDOUT, text=True, encoding="utf-8",
   830→                errors="replace", env=env, creationflags=NO_WINDOW)
   831→        except Exception as exc:
   832→            self.log("err", f"اجرای سرور ناموفق: {exc}")
   833→            self.set_status("اجرای سرور ناموفق", COLORS["err"])
   834→            return
   835→
   836→        self.log("ok", f"پروسهٔ سرور شروع شد (PID {self.proc.pid}) — منتظر آماده‌شدن…")
   837→        threading.Thread(target=self._pipe_logs, daemon=True).start()
   838→        threading.Thread(target=self._wait_until_up, daemon=True).start()
   839→
   840→    def _pipe_logs(self):
   841→        """خروجی زندهٔ سرور در لاگ لانچر (خطاها دیگر مخفی نیستند)"""
   842→        proc = self.proc
   843→        if not proc or not proc.stdout:
   844→            return
   845→        for line in proc.stdout:
   846→            if self.stop_flag:
   847→                break
   848→            self.post(lambda l=line: self.log("srv", l.rstrip()))
   849→        code = proc.wait()
   850→
   851→        def done():
   852→            if self.stop_flag:
   853→                return
   854→            if self.server_ready:
   855→                self.set_running(False)
   856→                self.log("warn", f"پروسهٔ سرور تمام شد (کد {code})")
   857→            else:
   858→                self.set_status("❌ سرور بالا نیامد — لاگ بالا را ببینید", COLORS["err"])
   859→                self.log("err", f"سرور قبل از آماده‌شدن بسته شد (کد {code})")
   860→                self.log("info", "راهنما: خروجی سرور در همین لاگ بالاست؛ "
   861→                         "برای جزئیات بیشتر این را در ترمینال اجرا کن:  python panel_server.py")
   862→        self.post(done)
   863→
   864→    def _wait_until_up(self):
   865→        for _ in range(180):                        # حداکثر ~۹۰ ثانیه
   866→            if self.stop_flag or self.server_ready:
   867→                return
   868→            if self.proc and self.proc.poll() is not None:
   869→                return                              # خطا در _pipe_logs گزارش می‌شود
   870→            if server_up(0.8):
   871→                self.post(self._on_server_up)
   872→                return
   873→            time.sleep(0.5)
   874→        if self.stop_flag or self.server_ready:
   875→            return
   876→
   877→        def fail():
   878→            self.set_status("سرور در ۹۰ ثانیه پاسخ نداد", COLORS["err"])
   879→            self.log("err", "سرور در ۹۰ ثانیه آماده نشد — چند نکته:")
   880→            self.log("info", "• لاگ بالا را ببین؛ اگر خطای ماژول است، «نصب پیش‌نیازها» را بزن")
   881→            self.log("info", "• اگر پنل قبلاً در مرورگر باز بوده، «🌐 باز کردن پنل» را امتحان کن")
   882→            self.log("info", "• شاید پورت 8765 اشغال است: با  set PANEL_PORT=8800  و اجرای مجدد امتحان کن")
   883→            self.log("info", "• راه نهایی: در ترمینال  python panel_server.py  را اجرا کن تا خطا کامل دیده شود")
   884→        self.post(fail)
   885→
   886→    def _on_server_up(self):
   887→        self.server_ready = True
   888→        self.log("ok", f"سرور آماده است ✓ — باز کردن پنل: {PANEL_URL}")
   889→        self.set_running(True)
   890→        self.after(500, lambda: webbrowser.open(PANEL_URL))
   891→
   892→    def stop_server(self):
   893→        if not (self.proc and self.proc.poll() is None):
   894→            self.log("info", "سرور در حال اجرا نیست")
   895→            self.set_running(False)
   896→            return
   897→        self.log("warn", "در حال توقف سرور (مرورگرهای باز هم بسته می‌شوند)…")
   898→        try:
   899→            if os.name == "nt":
   900→                subprocess.run(
   901→                    ["taskkill", "/F", "/T", "/PID", str(self.proc.pid)],
   902→                    capture_output=True, creationflags=NO_WINDOW)
   903→            else:
   904→                self.proc.terminate()
   905→        except Exception as exc:
   906→            self.log("err", f"توقف ناموفق: {exc}")
   907→        self.server_ready = False
   908→
   909→    def open_panel(self):
   910→        if server_up(0.8):
   911→            webbrowser.open(PANEL_URL)
   912→        else:
   913→            messagebox.showinfo("سرور خاموش است",
   914→                                "اول دکمهٔ «اجرای سرور» را بزن")

   918→        if self.proc and self.proc.poll() is None:
   919→            if messagebox.askyesno(
   920→                    "سرور در حال اجراست",
   921→                    "با بستن لانچر، سرور و پنل هم خاموش شوند؟",
   922→                    icon="question"):

   924→                self.stop_server()

   932→def _console_mode(argv: list) -> int | None:
   933→    """اگر یکی از سوییچ‌های کنسلی خواسته شده بود، اجرا و کد خروج را برگردان؛
   934→    اگر GUI لازم است None."""
   935→    if "--build-bundle" in argv:
   936→        include = "--no-browsers" not in argv
   937→        try:
   938→            print(build_offline_bundle(include, _console_log))
   939→            print("\n✅ بستهٔ آفلاین ساخته شد.")
   940→            return 0
   941→        except Exception as exc:
   942→            print(f"[err] {exc}")
   943→            return 1
   944→    if "--install-offline" in argv:
   945→        if not bundle_available():
   946→            print("[err] پوشهٔ offline_bundle کنار پروژه پیدا نشد.")
   947→            return 1
   948→        probs = compat_problems(read_bundle_info())
   949→        if probs:
   950→            for p in probs:
   951→                print(f"[err] {p}")
   952→            return 1
   953→        msg = disk_msg()
   954→        if msg:
   955→            print(f"[err] {msg}")
   956→            return 1
   957→        ok, _missing = install_from_bundle(_console_log)
   958→        return 0 if ok else 1
   959→    if argv and argv[0] in ("-h", "--help"):
   960→        print(__doc__)
   961→        return 0
   962→    return None
